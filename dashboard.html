FUNCTION TO RENDER USER DETAILS function setUserProfile(email) { const emailElement = document.getElementById('user-email-display'); const handleElement = document.getElementById('user-handle-display'); const avatarElement = document.getElementById('avatar-initials'); if (!email) return; // Set full email emailElement.innerText = email; // Clean special characters to generate handle (e.g. john.doe@gmail.com -> @johndoe) const namePart = email.split('@')[0].replace(/[^a-zA-Z0-9]/g, ''); handleElement.innerText = `@${namePart.toLowerCase()}`; // Set avatar initials (first 2 letters) avatarElement.innerText = namePart.substring(0, 2).toUpperCase(); } // 2. AUTOMATICALLY FETCH EMAIL ON PAGE LOAD window.addEventListener('DOMContentLoaded', () => { // Grab the saved email from browser storage const savedEmail = localStorage.getItem('userEmail'); if (savedEmail) { // User is logged in -> render their profile setUserProfile(savedEmail); } else { // Optional: Redirect to login page if no logged-in user is found alert('No active session found. Redirecting to login...'); window.location.href = 'index.html'; } }); // 3. PAYMENT BUTTON TRIGGER function triggerPayment(planName, amount) { const currentEmail = localStorage.getItem('userEmail') || 'Customer'; alert(`Initiating payment for ${planName} (₦${amount.toLocaleString()}) for ${currentEmail}...`); }
